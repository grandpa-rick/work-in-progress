\section*{1. The window is the Kodaira strip}
\subsection*{1.1 Theorem D (\gr{proved}; uses neither $(D)$ nor K0)}
Conventions: $C_n$ with $\alpha_i=e_i-e_{i+1}$ ($i<n$), $\alpha_n=2e_n$, $\omega_n=(1,\dots,1)$, and Demazure operators
$D_i f=(f-e^{-\alpha_i}s_if)/(1-e^{-\alpha_i})$. A subset $\kappa\subseteq[n]$ is the minimal coset representative
$w_\kappa$ with $\ell(w_\kappa)=|\kappa|$ (sum of elements); Bruhat order is top-aligned dominance. Fixed points
$L\leftrightarrow v_L$, under the matching $w_m=y_{n+1-m}$. Write
$D_{w_\kappa}(e^{t\omega_n})=\sum_{L\le\kappa}a^\kappa_L\,e^{tv_L}$.

\begin{thmD}
With $A_\kappa:=(-1)^{|\kappa|}\prod_j W_{\le\kappa_j}^{-1}$ and $W_{\le k}=w_1\cdots w_k$,
\[
\ell_\kappa(f)=A_\kappa\sum_L a^\kappa_L\,w_L^2\,f(L)=A_\kappa W_n\,\chi_T\big(X_\kappa,\ f\otimes L_{-\omega_n}\big).
\]
So the dual functional to $\{F_\lambda\}$ is integration over $X_\kappa\subset LG(n,2n)$.
\end{thmD}
\emph{Proof outline.} Put $g_\lambda:=w_M^2F_\lambda(M)$, and let $W$ act on functions of subsets by
$(s\cdot g)(L)=s(g(sL))$. Here $s_m$ swaps $w_m\leftrightarrow w_{m+1}$, and $\sigma=s_n$ sends $w_1\mapsto1/w_1$ and
$L\mapsto L\,\triangle\,\{1\}$. Then:
\begin{itemize}[leftmargin=*,itemsep=1pt]
\item \textbf{Lemma A$_m$.} $s_m g_\lambda=g_\lambda$ if $m{+}1\notin\lambda$, and
  $s_m g_\lambda=g_\lambda+(w_m-w_{m+1})g_{\lambda'}$ if $m{+}1\in\lambda\not\ni m$ (with $\lambda'$: $m{+}1\mapsto m$).
  Both are one column operation in the alternant of Lemma A.
\item \textbf{Lemma B.} With $z=w_1$: $\sigma g_\lambda=z^{-2(r+1)}g_\lambda$ if $1\notin\lambda$, and
  $\sigma g_\lambda=z^{-2r}g_\lambda-(z^2-1)z^{-2r-1}g_{\lambda\setminus1}$ if $1\in\lambda$. This one is a three-pole
  residue identity.
\end{itemize}
Demazure's recursion for $a^\kappa$ then gives $\Lambda_\kappa(g_\lambda)=\delta_{\kappa\lambda}/A_\kappa$ by induction
on $|\kappa|$, with two diagonal checks per step. Exact termwise check: 64/64, 256/256 (two points), 1024/1024 at
$n=3,4,5$. Two controls break it: the matching $w_m=y_m$ (25 mismatches), and $W_{<\kappa_j}$ in place of
$W_{\le\kappa_j}$ (34).
